Koneksi Auto-Update Google Sheets

Masukkan URL Google Apps Script Web App agar setiap kali menekan tombol Simpan, data langsung terkirim otomatis ke Google Spreadsheet online Anda.

// Tempel di Google Sheet -> Ekstensi -> Apps Script -> Deploy Web App (Anyone):

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["ID","Tanggal","Afdeling","Blok","Pekerjaan","Hasil","Satuan","HK","Bahan","Mandor","Keterangan"]);
  }
  sheet.appendRow([data.id, data.tanggal, data.afdeling, data.blok, data.pekerjaan, data.hasil, data.satuan, data.hk, data.bahan, data.mandor, data.keterangan]);
  return ContentService.createTextOutput(JSON.stringify({status:"ok"})).setMimeType(ContentService.MimeType.JSON);
}
